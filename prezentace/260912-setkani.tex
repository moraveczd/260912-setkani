\section{Wikipedie}
\frame{
	\frametitle{}
	\vfill

	\begin{itemize}
		\item Wikipedie je encyklopedie -- terciární zdroj informací.
		\item Je dostupná zcela zdarma, bez reklam.
		\item Může sloužit jako dobrý rozcestník na sekundární a primární zdroje informací k danému tématu.
		\item U každého článku jsou také zpravidla uvedeny knihy a další literární zdroje pojednávající o tématu.
		\item \textbf{Ukázka} -- \href{https://cs.wikipedia.org/wiki/Vod\%C3\%ADk}{\underline{Vodík}}
		\item Česká Wikipedie -- téměř 600 000 článků
		\item Anglická Wikipedie -- více než 7 200 000 článků
	\end{itemize}
	\vfill
}

\section{Wikimedia Commons}
\frame{
	\frametitle{}
	\vfill
	\begin{columns}
		\begin{column}{.8\textwidth}
			\begin{itemize}
				\item \href{https://commons.wikimedia.org/}{\underline{Wikimedia Commons}} je organizované centrální úložiště licenčně volných fotografií, videí, ilustrací, zvuků a dalšího multimediálního obsahu.
				\item V současnosti obsahuje více než 147 miliónů volně využitelných souborů.
				\item Fotografie a ilustrace jsou nezbytnou součástí výukových materiálů, proto může být projekt Wikimedia Commons zajímavý pro vyučující.
				\item Fotografie a obrázky jsou nejčastěji dostupné pod některou z variant licence \textit{Creative Commons} nebo \textit{Public Domain}. Přesný způsob
				\item \textbf{Ukázka} -- \href{https://is.muni.cz/www/moravec/c2062_anorganicka_chemie_ii/}{\underline{C2062 Anorganická chemie II}}
				\item \textbf{Ukázka} -- \href{https://commons.wikimedia.org/wiki/File:Potassium_chlorate_and_gummi_bear_experiment.webm}{\underline{Videozáznamy chemických pokusů}}
			\end{itemize}
		\end{column}

		\begin{column}{.25\textwidth}
